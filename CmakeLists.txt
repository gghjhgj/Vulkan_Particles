)

if(NOT SFML_FOUND)

        SFML
        GIT_REPOSITORY https://github.com/SFML/SFML.git
        GIT_TAG 3.0.1
        GIT_SHALLOW TRUE
    )

    set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_DOC OFF CACHE BOOL "" FORCE)

    FetchContent_MakeAvailable(SFML)

    message(STATUS "SFML found.")

endif()

    message(STATUS "Checking Dear ImGui")
    message(STATUS "================================================")

    find_package(imgui QUIET)

    if(NOT imgui_FOUND)

            add_library(imgui STATIC

                ${imgui_src_SOURCE_DIR}/imgui_widgets.cpp
            )

            target_include_directories(imgui
                PUBLIC
                    ${imgui_src_SOURCE_DIR}
            )

        endif()

            add_library(imgui_vulkan STATIC
                ${imgui_src_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
            )

            target_include_directories(imgui_vulkan
                PUBLIC
                    ${imgui_src_SOURCE_DIR}
                    ${imgui_src_SOURCE_DIR}/backends
            )

            target_link_libraries(imgui_vulkan
                PUBLIC
                    imgui

# ================================================================

    message(STATUS "")
    message(STATUS "================================================")
    message(STATUS "Checking mimalloc")
    message(STATUS "================================================")

    find_library(MIMALLOC_LIBRARY
        NAMES mimalloc
    )

    find_path(MIMALLOC_INCLUDE_DIR
        NAMES mimalloc.h
    )

    if(NOT MIMALLOC_LIBRARY OR NOT MIMALLOC_INCLUDE_DIR)

        message(STATUS
            "mimalloc not found. Downloading automatically..."
        )

        FetchContent_Declare(
            mimalloc
            GIT_REPOSITORY https://github.com/microsoft/mimalloc.git
            GIT_TAG v3.1.5
            GIT_SHALLOW TRUE
        )

        set(MI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(MI_BUILD_OBJECT OFF CACHE BOOL "" FORCE)
        set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)

        FetchContent_MakeAvailable(mimalloc)

        if(TARGET mimalloc-static)

            set(MIMALLOC_TARGET mimalloc-static)

        elseif(TARGET mimalloc)

            set(MIMALLOC_TARGET mimalloc)

        else()

            message(FATAL_ERROR
                "mimalloc was downloaded but no usable target was created."
            )

        endif()

        message(STATUS
            "mimalloc downloaded and configured."
        )

    else()

        add_library(mimalloc_external UNKNOWN IMPORTED)

        set_target_properties(mimalloc_external PROPERTIES
            IMPORTED_LOCATION
                "${MIMALLOC_LIBRARY}"

            INTERFACE_INCLUDE_DIRECTORIES
                "${MIMALLOC_INCLUDE_DIR}"
        )

        set(MIMALLOC_TARGET mimalloc_external)

        message(STATUS "mimalloc found.")

    endif()

endif()

# ================================================================
# SHADERS
# ================================================================

if(BUILD_SHADERS)

    message(STATUS "")
    message(STATUS "================================================")
    message(STATUS "Searching GLSL shaders")
    message(STATUS "================================================")

    file(GLOB SHADER_SOURCES
        CONFIGURE_DEPENDS
        "${PROJECT_SOURCE_DIR}/shaders/*.glsl"
    )

    set(SHADER_OUTPUTS)

    foreach(SHADER ${SHADER_SOURCES})

        get_filename_component(
            SHADER_NAME
            "${SHADER}"
            NAME
        )

        # ------------------------------------------------------------
        # Detect shader stage
        # ------------------------------------------------------------

        if(SHADER_NAME MATCHES "_vert\\.glsl$")

            set(SHADER_STAGE vertex)

        elseif(SHADER_NAME MATCHES "_frag\\.glsl$")

            set(SHADER_STAGE fragment)

        elseif(SHADER_NAME MATCHES "_comp\\.glsl$")

            set(SHADER_STAGE compute)

        else()

            message(WARNING
                "Unknown shader stage for ${SHADER_NAME}. "
                "Assuming compute."
            )

            set(SHADER_STAGE compute)

        endif()

        # ------------------------------------------------------------
        # Output filename
        # ------------------------------------------------------------

        string(
            REGEX REPLACE
            "\\.glsl$"
            ".spv"
            SHADER_OUTPUT_NAME
            "${SHADER_NAME}"
        )

        # ------------------------------------------------------------
        # BUILD output
        # ------------------------------------------------------------

        set(BUILD_SPV
            "${CMAKE_BINARY_DIR}/bin/shaders/${SHADER_OUTPUT_NAME}"
        )

        # ------------------------------------------------------------
        # SOURCE output
        #
        # Your C++ currently loads:
        #
        # shaders/particle_vert.spv
        #
        # so keep a copy there too.
        # ------------------------------------------------------------

        set(SOURCE_SPV
            "${PROJECT_SOURCE_DIR}/shaders/${SHADER_OUTPUT_NAME}"
        )

        # ------------------------------------------------------------
        # Compile GLSL -> SPIR-V
        # ------------------------------------------------------------

        add_custom_command(
            OUTPUT
                "${BUILD_SPV}"

            COMMAND
                ${CMAKE_COMMAND}
                -E make_directory
                "${CMAKE_BINARY_DIR}/bin/shaders"

            COMMAND
                "${GLSLC_EXECUTABLE}"
                "-fshader-stage=${SHADER_STAGE}"
                "${SHADER}"
                "-o"
                "${BUILD_SPV}"

            COMMAND
                ${CMAKE_COMMAND}
                -E copy_if_different
                "${BUILD_SPV}"
                "${SOURCE_SPV}"

            DEPENDS
                "${SHADER}"

            COMMENT
                "GLSLC ${SHADER_STAGE}: ${SHADER_NAME}"

            VERBATIM
        )

        list(APPEND
            SHADER_OUTPUTS
            "${BUILD_SPV}"
        )

        message(STATUS
            "  ${SHADER_NAME} -> ${SHADER_STAGE}"
        )

    endforeach()

    # ------------------------------------------------------------
    # Shader target
    # ------------------------------------------------------------

    add_custom_target(shader
        ALL
        DEPENDS
            ${SHADER_OUTPUTS}
    )

    # App waits for shaders.
    add_dependencies(app shader)

endif()
