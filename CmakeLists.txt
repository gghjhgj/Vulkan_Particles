if(USE_MIMALLOC)

    message(STATUS "")
    message(STATUS "================================================")
    message(STATUS "Checking mimalloc")
    message(STATUS "================================================")

    find_library(MIMALLOC_LIBRARY
        NAMES mimalloc
    )

    find_path(MIMALLOC_INCLUDE_DIR
        NAMES mimalloc.h
    )

    if(NOT MIMALLOC_LIBRARY OR NOT MIMALLOC_INCLUDE_DIR)

        message(STATUS
            "mimalloc not found. Downloading automatically..."
        )

        FetchContent_Declare(
            mimalloc
            GIT_REPOSITORY https://github.com/microsoft/mimalloc.git
            GIT_TAG v3.1.5
            GIT_SHALLOW TRUE
        )

        set(MI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(MI_BUILD_OBJECT OFF CACHE BOOL "" FORCE)
        set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)

        FetchContent_MakeAvailable(mimalloc)

        if(TARGET mimalloc-static)

            set(MIMALLOC_TARGET mimalloc-static)

        elseif(TARGET mimalloc)

            set(MIMALLOC_TARGET mimalloc)

        else()

            message(FATAL_ERROR
                "mimalloc was downloaded but no usable target was created."
            )

        endif()

        message(STATUS
            "mimalloc downloaded and configured."
        )

    else()

        add_library(mimalloc_external UNKNOWN IMPORTED)

        set_target_properties(mimalloc_external PROPERTIES
            IMPORTED_LOCATION
                "${MIMALLOC_LIBRARY}"

            INTERFACE_INCLUDE_DIRECTORIES
                "${MIMALLOC_INCLUDE_DIR}"
        )

        set(MIMALLOC_TARGET mimalloc_external)

        message(STATUS "mimalloc found.")

    endif()

endif()

# ================================================================
# SHADERS
# ================================================================

if(BUILD_SHADERS)

    message(STATUS "")
    message(STATUS "================================================")
    message(STATUS "Searching GLSL shaders")
    message(STATUS "================================================")

    # Szukanie wszystkich plikow z typowymi rozszerzeniami shaderow
    file(GLOB_RECURSE SHADER_SOURCES
        CONFIGURE_DEPENDS
        "${PROJECT_SOURCE_DIR}/shaders/*.glsl"
        "${PROJECT_SOURCE_DIR}/shaders/*.vert"
        "${PROJECT_SOURCE_DIR}/shaders/*.frag"
        "${PROJECT_SOURCE_DIR}/shaders/*.comp"
        "${PROJECT_SOURCE_DIR}/shaders/*.geom"
        "${PROJECT_SOURCE_DIR}/shaders/*.tesc"
        "${PROJECT_SOURCE_DIR}/shaders/*.tese"
        "${PROJECT_SOURCE_DIR}/shaders/*.rgen"
        "${PROJECT_SOURCE_DIR}/shaders/*.rmiss"
        "${PROJECT_SOURCE_DIR}/shaders/*.rchit"
        "${PROJECT_SOURCE_DIR}/shaders/*.rahit"
    )

    set(SHADER_OUTPUTS)

    foreach(SHADER ${SHADER_SOURCES})

        get_filename_component(
            SHADER_NAME
            "${SHADER}"
            NAME
        )

        # ------------------------------------------------------------
        # Detect shader stage (obsluguje np. fluid.comp, fluid.comp.glsl, fluid_comp.glsl)
        # ------------------------------------------------------------

        if(SHADER_NAME MATCHES "[._]vert(\\.[^.]*)?$")

            set(SHADER_STAGE vertex)

        elseif(SHADER_NAME MATCHES "[._]frag(\\.[^.]*)?$")

            set(SHADER_STAGE fragment)

        elseif(SHADER_NAME MATCHES "[._]comp(\\.[^.]*)?$")

            set(SHADER_STAGE compute)

        elseif(SHADER_NAME MATCHES "[._]geom(\\.[^.]*)?$")

            set(SHADER_STAGE geometry)

        elseif(SHADER_NAME MATCHES "[._]tesc(\\.[^.]*)?$")

            set(SHADER_STAGE tesscontrol)

        elseif(SHADER_NAME MATCHES "[._]tese(\\.[^.]*)?$")

            set(SHADER_STAGE tesseval)

        elseif(SHADER_NAME MATCHES "[._]rgen(\\.[^.]*)?$")

            set(SHADER_STAGE rgen)

        elseif(SHADER_NAME MATCHES "[._]rmiss(\\.[^.]*)?$")

            set(SHADER_STAGE rmiss)

        elseif(SHADER_NAME MATCHES "[._]rchit(\\.[^.]*)?$")

            set(SHADER_STAGE rchit)

        elseif(SHADER_NAME MATCHES "[._]rahit(\\.[^.]*)?$")

            set(SHADER_STAGE rahit)

        else()

            # Pliki bez okreslonego etapu (np. particle.glsl, struct.glsl) sa pomijane
            continue()

        endif()

        # ------------------------------------------------------------
        # Output filename & relative paths
        # ------------------------------------------------------------

        file(RELATIVE_PATH
            REL_PATH
            "${PROJECT_SOURCE_DIR}/shaders"
            "${SHADER}"
        )

        # Jesli plik konczy sie na .glsl, podmieniamy .glsl na .spv (np. fluid.comp.glsl -> fluid.comp.spv)
        # Jesli konczy sie na np. .comp, dopisujemy .spv (np. fluid.comp -> fluid.comp.spv)
        if(REL_PATH MATCHES "\\.glsl$")
            string(
                REGEX REPLACE
                "\\.glsl$"
                ".spv"
                REL_SPV_PATH
                "${REL_PATH}"
            )
        else()
            set(REL_SPV_PATH "${REL_PATH}.spv")
        endif()

        set(BUILD_SPV
            "${CMAKE_BINARY_DIR}/bin/shaders/${REL_SPV_PATH}"
        )

        set(SOURCE_SPV
            "${PROJECT_SOURCE_DIR}/shaders/${REL_SPV_PATH}"
        )

        get_filename_component(
            BUILD_SPV_DIR
            "${BUILD_SPV}"
            DIRECTORY
        )

        # ------------------------------------------------------------
        # Compile GLSL -> SPIR-V
        # ------------------------------------------------------------

        add_custom_command(
            OUTPUT
                "${BUILD_SPV}"

            COMMAND
                ${CMAKE_COMMAND}
                -E make_directory
                "${BUILD_SPV_DIR}"

            COMMAND
                "${GLSLC_EXECUTABLE}"
                "-fshader-stage=${SHADER_STAGE}"
                "-I" "${PROJECT_SOURCE_DIR}/shaders"
                "${SHADER}"
                "-o"
                "${BUILD_SPV}"

            COMMAND
                ${CMAKE_COMMAND}
                -E copy_if_different
                "${BUILD_SPV}"
                "${SOURCE_SPV}"

            DEPENDS
                "${SHADER}"

            COMMENT
                "GLSLC ${SHADER_STAGE}: ${REL_PATH}"

            VERBATIM
        )

        list(APPEND
            SHADER_OUTPUTS
            "${BUILD_SPV}"
        )

        message(STATUS
            "  ${REL_PATH} -> ${SHADER_STAGE} (${REL_SPV_PATH})"
        )

    endforeach()

    # ------------------------------------------------------------
    # Shader target
    # ------------------------------------------------------------

    if(SHADER_OUTPUTS)
        add_custom_target(shader
            ALL
            DEPENDS
                ${SHADER_OUTPUTS}
        )

        add_dependencies(app shader)
    endif()

endif()
